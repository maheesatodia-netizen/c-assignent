1. Flipkart Product :


char productName[ ] = "Watch";

double rating = 4.5;

2. Zomato GST :

const float GST = 18.0;
float price = 500;
float finalPrice = price + (price * GST / 100);

printf("Final Price: %.2f", finalPrice);



3. Spotify Playlist :

char playlist[ ] = "My Favorites";
int songs = 20;
float duration = 3.5;

printf("My playlist %s has %d songs with an average duration of %.1f minutes.", playlist, songs, duration);
